## 1. Migration 003 — schema (authoritative)

```sql
-- candidate_profiles — 1:1 with auth.users
CREATE TABLE IF NOT EXISTS candidate_profiles (
  user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  display_name TEXT,
  headline TEXT,           -- e.g. "ML Engineer · 3y · Berlin"
  bio TEXT,
  skills TEXT[] DEFAULT '{}',
  languages TEXT[] DEFAULT '{}',          -- e.g. {en,zh,de}
  preferred_locations TEXT[] DEFAULT '{}',
  preferred_fields TEXT[] DEFAULT '{}',
  visible BOOLEAN DEFAULT FALSE,          -- opt-in to recruiter pool (FR-E5-1)
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- saved_jobs — candidate bookmarks (unique per user+job)
CREATE TABLE IF NOT EXISTS saved_jobs (
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  job_id TEXT REFERENCES jobs(id) ON DELETE CASCADE,
  saved_at TIMESTAMPTZ DEFAULT NOW(),
  PRIMARY KEY (user_id, job_id)
);

-- saved_filters — named filter sets for alerts
CREATE TABLE IF NOT EXISTS saved_filters (
  id SERIAL PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  name TEXT NOT NULL,                     -- "Berlin AI · Chinese required"
  filters JSONB NOT NULL,                 -- {field, location, languageLevel, q, remote, visa}
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- applications — per saved job, per user
CREATE TABLE IF NOT EXISTS applications (
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  job_id TEXT REFERENCES jobs(id) ON DELETE CASCADE,
  status TEXT DEFAULT 'saved' CHECK (status IN ('saved','applied','screening','interview','offer','rejected')),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  PRIMARY KEY (user_id, job_id)
);

-- cvs — metadata for Storage objects (bucket `cvs` private)
CREATE TABLE IF NOT EXISTS cvs (
  id SERIAL PRIMARY KEY,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  storage_path TEXT NOT NULL,             -- cvs/{user_id}/{uuid}.pdf
  file_name TEXT NOT NULL,
  file_size INT NOT NULL,
  mime_type TEXT NOT NULL,
  uploaded_at TIMESTAMPTZ DEFAULT NOW()
);
-- Bucket `cvs` created via Storage API or SQL: insert into storage.buckets (id,name,public) values ('cvs','cvs',false) on conflict do nothing;
-- RLS: enable on all 5 tables; policies: owner-only SELECT/INSERT/UPDATE/DELETE for authenticated where auth.uid()=user_id; service_role permissive. jobs public read already covers job existence checks. Storage object policies: authenticated can INSERT/SELECT own prefix cvs/{uid}/*, service_role all.
```

Idempotent via `IF NOT EXISTS`, `DROP POLICY IF EXISTS` then `CREATE POLICY`.
